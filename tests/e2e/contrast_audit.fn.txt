// Browser-side function (evaluated by Playwright, not imported). Returns text elements below WCAG AA contrast (4.5:1; 3:1 for large text). Gradients and maps are skipped.
() => {
  const rgba = (c) => { const m = c.match(/[\d.]+/g).map(Number); return { r: m[0], g: m[1], b: m[2], a: m.length > 3 ? m[3] : 1 }; };
  const lin = (v) => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); };
  const L = (c) => 0.2126 * lin(c.r) + 0.7152 * lin(c.g) + 0.0722 * lin(c.b);
  const blend = (top, bottom) => ({ r: top.r * top.a + bottom.r * (1 - top.a), g: top.g * top.a + bottom.g * (1 - top.a), b: top.b * top.a + bottom.b * (1 - top.a), a: 1 });
  const bgOf = (el) => { const stack = []; for (let n = el; n; n = n.parentElement) { const cs = getComputedStyle(n); if (cs.backgroundImage !== 'none') return null; const c = rgba(cs.backgroundColor); if (c.a > 0) stack.push(c); if (c.a >= 1) break; } let base = { r: 8, g: 12, b: 16, a: 1 }; for (const c of stack.reverse()) base = blend(c, base); return base; };
  const out = [];
  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  const seen = new Set();
  while (walker.nextNode()) {
    const t = walker.currentNode; if (!t.textContent.trim()) continue;
    const el = t.parentElement; if (!el || seen.has(el)) continue; seen.add(el);
    const cs = getComputedStyle(el); const r = el.getBoundingClientRect();
    if (r.width < 2 || r.height < 2 || cs.visibility === 'hidden' || cs.display === 'none' || el.closest('.leaflet-container,svg,script,style,[disabled],[aria-hidden="true"]')) continue;
    const bg = bgOf(el); if (!bg) continue;
    let fg = rgba(cs.color); fg = blend(fg, bg); const op = parseFloat(cs.opacity);
    const ratio = (Math.max(L(fg), L(bg)) + 0.05) / (Math.min(L(fg), L(bg)) + 0.05);
    const size = parseFloat(cs.fontSize), bold = parseInt(cs.fontWeight) >= 700;
    const need = (size >= 24 || (size >= 18.66 && bold)) ? 3 : 4.5;
    if (ratio < need) out.push({ ratio: +ratio.toFixed(2), need, text: t.textContent.trim().slice(0, 48), cls: (el.className && el.className.toString().slice(0, 50)) || el.tagName, opacity: op });
  }
  return out;
}
